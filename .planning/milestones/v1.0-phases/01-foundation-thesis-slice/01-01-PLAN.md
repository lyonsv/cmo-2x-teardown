site/js/data.js (ES module, no imports):
- export const GRADES = ['Benchmark','Modelled','Speculative']   // strongest first
- export const FLAGS = ['vendor-claimed','company-reported']
- export const SCENARIO = { multiplier: 2 }                       // the claim under test, not graded
- export const sources = { '<key>': { title, publisher, published, accessed, url, bias, read_via } }
    register row id for a source is 'src-<key>'
- export const assumptions = { 'asm-<name>': { id, label, kind: 'input'|'model'|'context'|'evidence', key, value, unit, display, grade, flag, basis } }
    value is a number, or { baseline, scaled } for scenario-specific model assumptions, or null for kind 'evidence'
    key is the calc.js field name (null for 'context' and 'evidence')
- export const claims = { '<id>': { id, value, unit, display, grade, flag, ref, derived?, decimals? } }
    ref = register row id the chip links to ('src-...' or 'asm-...')
    derived = dotted path into compute() output, e.g. 'baseline.output_per_head_per_quarter'
    decimals = rounding applied before comparison with compute()
- export function defaultInputs()      -> { campaigns_per_year, concepts_per_campaign, variants_per_concept, marketing_fte, reviewer_fte, multiplier }
- export function defaultAssumptions() -> { [key]: value } built from assumptions with kind 'model'
Rules: claim ids and assumption ids never collide; a data-fig id may name a claim OR an assumption (assumption chip href is '#'+its own id).

site/js/calc.js (pure ES module, no DOM, no imports from data.js):
- export function compute(inputs, assumptions) ->
    { ok: true, baseline: {...}, scenario: {...}, ratios: {...} } | { ok: false, errors: [{ field, message }] }
  Phase 1 skeleton fields: baseline/scenario.{ concepts, variants_per_concept, variants_per_year, output_per_head_per_quarter }, ratios.output_per_head
  Formulas (m = 1 for baseline, m = inputs.multiplier for scenario, k = assumptions.concept_growth_exponent):
    concepts = campaigns_per_year * concepts_per_campaign * m^k
    variants_per_concept = variants_per_concept * m^(1-k)
    variants_per_year = concepts * variants_per_concept
    output_per_head_per_quarter = variants_per_year / 4 / marketing_fte
  Never throws, never returns NaN or Infinity.

HTML markup contract (regex-checked; keep on one line each):
- number:  <span data-fig="ID">DISPLAY</span>
- chip:    <a class="chip c-b|c-m|c-s" href="#src-KEY|#asm-NAME">Benchmark|Modelled|Speculative</a>
- flag:    <span class="flag">vendor-claimed|company-reported</span>   (only directly after a chip)
- register row: <tr id="src-KEY"> / <tr id="asm-NAME"> inside <section id="s15">
- section: <section id="sNN" aria-labelledby="sNN-h"> with <div class="sec-head"><span class="sec-num">NN</span><h2 id="sNN-h">Title</h2></div>
